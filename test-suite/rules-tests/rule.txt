rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function schoolUser() {
      return request.auth != null
        && request.auth.token.email is string
        && request.auth.token.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$')
        // 2026-07-03 補修：要求 email_verified == true，防止透過 Email/Password
        // 自助註冊、拿一個尚未被任何 provider 使用過的校內信箱字串（例如老師已
        // 建立 studentBindings 但學生本人尚未第一次登入的空窗期）取得可通過
        // schoolUser() 的 token，進而讀取/竄改該生 binding 或冒用其座號建立月記。
        // Google 登入的真實使用者本來就是 email_verified=true，不受影響；
        // Email/Password 帳號需要真的收到並點擊驗證信才會變 true，等於要求
        // 「真的掌控該信箱」，把這條路徑拉回跟 Google OAuth 一樣「網域控制力
        // 歸屬校方」的狀態。⚠️ 部署前必須先確認 test-teacher@ / test-student@
        // 兩組測試帳號在 Firebase Console 的 email_verified 狀態為 true，否則會
        // 連 CI 自動登入一起擋下。
        && request.auth.token.email_verified == true;
    }
    function emailKey() {
      return request.auth.token.email
        .replace('@', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_');
    }
    function isAdmin() {
      return schoolUser()
        && (
          exists(/databases/$(database)/documents/admins/$(request.auth.uid))
          || exists(/databases/$(database)/documents/admins/$(emailKey()))
        );
    }
    function validAdminWrite() {
      return request.resource.data.email is string
        && request.resource.data.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$')
        && (
          !request.resource.data.keys().hasAny(['protected'])
          || request.resource.data.protected is bool
        );
    }
    function keepsProtectedFlag() {
      return !resource.data.keys().hasAny(['protected'])
        || resource.data.protected != true
        || (
          request.resource.data.keys().hasAny(['protected'])
          && request.resource.data.protected == true
        );
    }
    // 2026-07 補修（稽核 #8）：fcmTokens 文件寫入原本完全沒有欄位型別/大小驗證，
    // 前端（student.html／teacher.html 的 initPushNotifications()）固定只會寫入
    // createdAt（ISO 字串）與截斷至 200 字的 userAgent（JS 端截斷，見
    // AI_推播系統說明.md 3.1 節），但因規則本身沒有驗證，理論上使用者可直接呼叫
    // Firestore API，在自己的 uid/adminId 路徑下寫入任意型別或超長字串。攻擊範圍天生
    // 侷限在寫入者自己的裝置清單（get/list 皆固定 false，讀取只透過 Admin SDK），
    // 不會波及他人，這裡純粹是讓規則本身也能擋下這類寫入，不只依賴前端行為維持乾淨。
    // hasOnly() 同時鎖住欄位清單，避免夾帶其他未預期欄位。
    function validFcmTokenWrite() {
      return request.resource.data.keys().hasOnly(['createdAt', 'userAgent'])
        && request.resource.data.createdAt is string
        && request.resource.data.userAgent is string
        && request.resource.data.userAgent.size() <= 200;
    }
    // 2026-07-25 補修：entriesCompleteAt（「篇數真正達到當月最少應繳篇數 minEntries 那一刻」
    // 的時間戳，2026-07-24 由 student.html saveJournal() 新增寫入，供 teacher.html
    // isJournalLate() 判斷遲交用，取代原本只看 submittedAt 的做法）原本完全沒有規則層
    // 驗證。跟 teacherComment 等老師專屬欄位不同，這個欄位在 CREATE 當下不是永遠固定為
    // null——若學生第一次繳交就直接寫滿達標篇數，entriesCompleteAt 在 CREATE 當下就會是
    // 一個真正的時間戳，所以不能要求「一律為 null」，只能做格式驗證：值必須是 null，或是
    // 符合 localISOStr() 輸出格式的字串。Firestore Rules 無法把任意字串解析成 Timestamp
    // 去跟 request.time 比較範圍（跟 studentReplyContentAt 那次遇到的語言限制相同），
    // 格式驗證沒辦法阻止學生把這個欄位偽造成「格式合法但內容是假的」過去日期讓自己看起來
    // 準時（isJournalLate() 目前用字串字典序比較 entriesCompleteAt 跟 closeDate，內容是否
    // 誠實無法在規則層驗證，屬已知殘留風險，比照 submittedAt 現況接受）。但這條格式驗證
    // 仍值得加：若完全不驗證，空字串或任何短字串在字典序比較下都可能意外小於 closeDate
    // 而被誤判為「準時」，等於不需要偽造任何看起來像日期的內容就能免費騙過遲交判斷；加上
    // 格式驗證後，至少必須提供一個格式正確的日期字串。刻意不另外加 .size() 上限（不像
    // studentReplyAt 那樣）——`^...$` 完整錨定頭尾的格式正則本身已經把字串長度鎖死在固定
    // 19 字元（4+1+2+1+2+1+2+1+2+1+2），任何通過這個正則的字串長度必然等於19，另外加長度
    // 上限只是重複同一件事、不會多擋下任何情況。抽成獨立函式（而非像原本草稿那樣在
    // CREATE／UPDATE 兩處各寫一次相同的邏輯）跟 validFcmTokenWrite() 同一個理由：避免
    // 兩處各自演化又脫鉤，且讓 check-rule-diff.js 的 findUnmonitoredFunctions() 自我檢查
    // 機制能夠監控到這個函式本身（已同步在 RISK_TARGETS 補上對應條目）。
    function validEntriesCompleteAt() {
      return request.resource.data.get('entriesCompleteAt', null) == null || (
        request.resource.data.entriesCompleteAt is string
        && request.resource.data.entriesCompleteAt.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}$')
      );
    }
    // 2026-08-17 新增，2026-08-18 補修：entriesFirstCompleteAt——修正 entriesCompleteAt 的
    // 一個邊界案例。學生先在期限內達標（entriesCompleteAt 記下當時時間），之後某次編輯
    // 刪除一則工作摘要導致篇數掉回未達標（entriesCompleteAt 依既有設計正確清空為 null，
    // 這個行為完全不變，不修改 validEntriesCompleteAt() 或 entriesCompleteAt 本身），
    // 之後補寫回篇數達標時，entriesCompleteAt 會被視為「這次才剛好跨過門檻」，改記錄
    // 補寫當下（可能已逾期）的時間，讓明明期限內就已經誠實達標過一次的學生被誤判遲交。
    // 這個情境源自使用者提問後發現，跟既有文件記載的「規則層無法驗證日期誠實性」殘留
    // 風險不是同一類——那個是防造假，這個是誠實操作也會被系統自己的計算邏輯搞錯，完整
    // 背景見 AI_CONTEXT_歷程.md 對應討論。entriesCompleteAt 本身不受影響，teacher.html
    // 的 isJournalLate() 改為優先讀這個新欄位，讀不到才退回原本的
    // entriesCompleteAt || submittedAt。
    // 2026-08-18 補修背景：原始設計「一旦有值就不可逆保留、不論篇數怎麼增減都不會被覆蓋
    // 或清空」忽略了 minEntries 本身也可能被老師事後調高——若調高後學生才在新門檻下補齊
    // 篇數，這個欄位仍然會凍結在舊門檻下就已經記錄的舊日期，讓 isJournalLate() 誤判為
    // 準時。跟原始設計要修的「篇數中途掉回不足又補回來」在既有輸入（篇數前後值＋目前
    // 門檻）下訊號完全相同，無法只靠調整判斷式區分，需要新增
    // entriesFirstCompleteAtCount（凍結當下實際驗證過的篇數）才能讓 student.html
    // computeEntriesFirstCompleteAt() 在每次存檔時重新檢查凍結是否還撐得住目前門檻，
    // 完整計算邏輯見該函式上方註解。這裡同步把格式驗證擴充成同時驗證這個新欄位——兩者
    // 語意緊密耦合（entriesFirstCompleteAt 為 null 時 entriesFirstCompleteAtCount 也必須
    // 為 null，非 null 時必須同時是合法格式），合併成一個函式驗證，避免各自演化脫鉤，
    // 跟 studentReplyContentAt == studentReplyAt 一致性要求是同一種考量。
    function validEntriesFirstCompleteAt() {
      return (
        request.resource.data.get('entriesFirstCompleteAt', null) == null
        && request.resource.data.get('entriesFirstCompleteAtCount', null) == null
      ) || (
        request.resource.data.entriesFirstCompleteAt is string
        && request.resource.data.entriesFirstCompleteAt.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9]{2}:[0-9]{2}:[0-9]{2}$')
        && request.resource.data.get('entriesFirstCompleteAtCount', -1) is int
        && request.resource.data.entriesFirstCompleteAtCount >= 0
      );
    }
    // 2026-08-18 補修：原本「一旦非 null 就必須完全相等」的強不可逆保護，隨
    // entriesFirstCompleteAt 改為「可在 entriesFirstCompleteAtCount 撐不住目前 minEntries
    // 時合法失效重算」後不再適用——失效重算後的新值本來就會跟舊值不同，強制完全相等會
    // 把合法的失效重算也一併擋下。改成較弱但仍然有意義的保護：只要求新值不得比舊值
    // 「更早」（放寬成非遞減，不是維持不變）。這樣做仍然安全的理由：目前設計下唯一合法
    // 的改變路徑（student.html computeEntriesFirstCompleteAt() 第③分支）永遠是把值改成
    // 這次存檔當下的時間，一定晚於任何舊的歷史日期，所有合法情境天生就滿足非遞減；這條
    // 規則堵住的是技術使用者直接呼叫 API，把已經凍結的日期竄改成比目前紀錄「更早」的
    // 偽造日期（原本這個函式要防的攻擊路徑，見下方殘留風險說明）。字串比較（>=）在這裡
    // 是安全的：兩邊皆已經過上面 validEntriesFirstCompleteAt() 的固定19字元格式驗證，
    // ISO 格式（YYYY-MM-DDTHH:mm:ss，零補齊、固定寬度）字典序等同時間先後順序，跟
    // teacher.html isJournalLate() 現有的 entriesCompleteAt vs closeDate 字典序比較是
    // 同一種手法。
    // ⚠️ 已知殘留風險（跟使用者確認過的取捨，非遺漏）：這裡不驗證「這次改變是否真的因為
    // minEntries 被調高才合法觸發」——要做到這一層驗證，規則需要額外 get() 查
    // /deadlines/{semester}-{month} 取得目前 minEntries、在規則層重算一次
    // resolveMinEntries() 的 fallback 邏輯才能判斷，換來的是每次月記寫入多一次 Firestore
    // 讀取、規則邏輯明顯變複雜，且 resolveMinEntries() 的 fallback 邏輯會同時存在
    // student.html 跟 rule.txt 兩份、未來要記得同步（跟陷阱17／29「兩份需要人工同步的
    // 複本」是同一種風險）。技術使用者仍可能謊稱「因為門檻提高」而正常寫入一個較晚但
    // 捏造的新日期＋灌高 entriesFirstCompleteAtCount，這跟 entriesCompleteAt／
    // submittedAt 現有的「規則層無法驗證日期誠實性」殘留風險是同一等級，屬刻意選擇後的
    // 已知取捨。
    function keepsEntriesFirstCompleteAtOnceSet() {
      return resource.data.get('entriesFirstCompleteAt', null) == null
        || request.resource.data.get('entriesFirstCompleteAt', null) >= resource.data.entriesFirstCompleteAt;
    }
    // 2026-08-13 補修：稽核發現 salaryPhotos（2026-08-05 隨「薪資單可上傳多張」功能新增）
    // 只在前端 student.html 檢查數量上限（SALARY_PHOTO_MAX_COUNT=5）與總大小上限
    // （SALARY_PHOTO_TOTAL_LIMIT=650000字元），rule.txt 完全沒有對應驗證——技術使用者
    // 可繞過 UI 直接呼叫 API，寫入任意數量、任意大小的字串陣列到自己的月記文件。受
    // ownerUid 保護，只影響學生自己這份文件（撐爆逼近 Firestore 單文件 1MiB 上限，
    // 屆時連自己這份月記的正常存檔都會開始失敗），非越權或資料外洩，屬中風險而非高風險，
    // 跟 fcmTokens.userAgent（validFcmTokenWrite()）、studentReplyAt 先前補上 .size()
    // 上限是同一類問題。只鎖陣列長度（對應前端 5 張上限），不逐張做位元組層級的大小
    // 驗證——Firestore Rules 沒辦法有效率地對變動長度陣列逐項驗證位元組大小，跟 entries[]
    // 既有的語言限制是同一種取捨，不勉強做。用 .get(key, []) 帶預設值，欄位不存在時視為
    // 空陣列（通過），欄位存在但型別不是 list 時 `is list` 直接評估為 false（擋下），
    // 同時涵蓋「型別錯誤」與「陣列過長」兩種情況。
    function validSalaryPhotos() {
      return request.resource.data.get('salaryPhotos', []) is list
        && request.resource.data.get('salaryPhotos', []).size() <= 5;
    }
    // 2026-08-20 補修：月記的文件 ID 是 student.html saveJournal() 唯一的定位鍵，格式為
    // {seatNo}-{semester}-{month}。老師端統計卻讀資料欄位的 semester/month 分組；若只在
    // 前端組 ID、規則沒有同步驗證，技術使用者可直接呼叫 Firestore API，讓文件 ID 與欄位
    // 不一致，進而把同一份月記算進錯誤月份或學期。月份逐一比對而非使用 string(month)，是
    // 為了不依賴 Rules 語言的數字轉字串轉型，並同時把 month 限定為實際月份 1~12。
    function hasMatchingJournalId(journalId) {
      return request.resource.data.seatNo is string
        && request.resource.data.semester is string
        && request.resource.data.month is int
        && (
          (request.resource.data.month == 1 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-1')
          || (request.resource.data.month == 2 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-2')
          || (request.resource.data.month == 3 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-3')
          || (request.resource.data.month == 4 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-4')
          || (request.resource.data.month == 5 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-5')
          || (request.resource.data.month == 6 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-6')
          || (request.resource.data.month == 7 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-7')
          || (request.resource.data.month == 8 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-8')
          || (request.resource.data.month == 9 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-9')
          || (request.resource.data.month == 10 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-10')
          || (request.resource.data.month == 11 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-11')
          || (request.resource.data.month == 12 && journalId == request.resource.data.seatNo + '-' + request.resource.data.semester + '-12')
        );
    }
    // 2026-08-26 新增：/students/{docId} 原本用 docId.matches('.*_' + seatNo) 把 seatNo
    // 字串直接拼接進正則表達式（.matches() 的參數是 regex，見 rules.String 文件）。座號
    // 目前固定是純數字（如 "01"~"45"），不含任何正則特殊字元，正常情況不會被觸發；但
    // rule.txt 本身沒有在任何地方強制驗證座號格式必須是 [0-9]+，如果座號格式未來改版
    // 不慎引入正則特殊字元（例如 "." "*" "(" 等），.matches() 就會把它當成正則語法而非
    // 字面字元解讀，導致比對範圍意外放寬（例如座號 "0.1" 會連座號 "0A1" 都比對成功，因為
    // 正則的 "." 代表任意字元）。改用 split('_') 取最後一段做精確字串相等比較，完全避開
    // 把使用者資料當成正則語法解讀的風險，不需要仰賴「座號格式永遠是純數字」這個假設。
    function matchesOwnSeatNo(docId) {
      let seatNo = get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo;
      let parts = docId.split('_');
      return docId == seatNo || parts[parts.size() - 1] == seatNo;
    }
    match /admins/{adminId} {
      allow get: if schoolUser() && (adminId == request.auth.uid || adminId == emailKey() || isAdmin());
      allow list: if isAdmin();
      allow create: if isAdmin() && validAdminWrite();
      allow update: if isAdmin() && validAdminWrite() && keepsProtectedFlag();
      allow delete: if isAdmin()
        && (
          !resource.data.keys().hasAny(['protected'])
          || resource.data.protected != true
        );
      // 2026-07 新增：老師/管理員端 Web Push 用的 FCM registration token。
      // 比照 /admins/{adminId} 本身「uid 或 emailKey 皆可」的判斷方式，只有本人能寫入/刪除自己的 token；
      // 讀取只透過伺服器端腳本的 Admin SDK（notify-service/send-push-notifications.js，由
      // GitHub Actions 排程執行，非 Cloud Functions；Admin SDK 本來就會繞過此規則），
      // 前端不需要也不應該讀回自己的 token 清單。
      match /fcmTokens/{tokenId} {
        // 2026-07 補修（稽核 #8）：create/update 額外套用 validFcmTokenWrite() 做欄位
        // 型別/大小驗證；delete 刻意分開一條、不呼叫 validFcmTokenWrite()——delete 時
        // request.resource 為 null，呼叫依賴 request.resource.data 的函式會直接評估為
        // 不通過，等同誤擋刪除本身。
        allow create, update: if isAdmin() && (adminId == request.auth.uid || adminId == emailKey()) && validFcmTokenWrite();
        allow delete: if isAdmin() && (adminId == request.auth.uid || adminId == emailKey());
        allow get, list: if false;
      }
    }
    match /studentBindings/{bindingId} {
      // 2026-07-03 補修：原本只比對 bindingId == emailKey()，完全沒呼叫 schoolUser()，
      // 導致同一批新增的 email_verified 檢查沒套用到這裡——只要 email 字串對得上
      // 對應的 emailKey()，不論這個 token 是否真的驗證過都能讀到該筆 binding
      // （老師開學建好名冊、學生本人尚未第一次登入的空窗期正是會被利用的情境）。
      // 補上 schoolUser() 後，讀取者必須先通過「校內信箱格式 + email_verified==true」
      // 才會進一步比對 bindingId，其餘條件（bindingId 對應 emailKey()）不變。
      allow get: if (schoolUser() && bindingId == emailKey()) || isAdmin();
      allow list, create, delete: if isAdmin();
      // 2026-08-26 補修：原本 UPDATE 只比對 resource.data.email == request.auth.token.email，
      // 沒有像 get 規則一樣同時要求 bindingId == emailKey()——等於完全信任「這筆文件的
      // email 欄位是對的」，而不是獨立確認「這筆文件本來就是這個人的」。若曾經有任兩筆
      // studentBindings 文件的 email 欄位因為老師端輸入疏失（複製貼上沒改、跨學期名冊
      // migration 誤植）變成同一個值，該真人就能把自己的 uid 綁到「bindingId 根本不是
      // 自己 emailKey()」的另一筆文件上，等同劫持不屬於自己的座號綁定。補回跟 get 規則
      // 一致的 bindingId == emailKey() 檢查後，即使 email 欄位重複，也只有 bindingId 本身

            && request.resource.data.get('journalSubmitNotifiedAt', null) == null

            // 2026-08-17 補修，2026-08-18 補修：entriesFirstCompleteAt（＋2026-08-18新增的

            // validEntriesFirstCompleteAt()（定義於檔案前段、validEntriesCompleteAt() 旁，
            // 完整理由見該函式上方註解）。CREATE 當下沒有「舊值」可比較，只做格式驗證，
            // 「一旦凍結、門檻沒被調高過就不能倒退」是 UPDATE 分支才有意義的限制，見下方
            // update 對應說明。
            && validEntriesFirstCompleteAt()
            // 2026-08-13 補修：salaryPhotos 陣列長度驗證（見上方 validSalaryPhotos() 說明）
            && validSalaryPhotos()
          ) || isAdmin();
        allow update: if (
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.ownerUid == resource.data.ownerUid
            && request.resource.data.ownerEmail == resource.data.ownerEmail
            && request.resource.data.storagePath == 'user'
            // 2026-06-27 補修：一般編輯不可變更 seatNo（建立時已驗證過，編輯時維持不變即可，
            // 不重新查 studentBindings——避免老師重新分配座號這種正常操作意外鎖死舊月記的編輯權）
            && request.resource.data.seatNo == resource.data.seatNo
            && hasMatchingJournalId(journalId)
            && request.resource.data.get('teacherComment', null) == null
            && request.resource.data.get('teacherReviewed', false) == false
            && request.resource.data.get('reviewedAt', null) == null
            && request.resource.data.get('teacherCommentUnread', false) == false
            && request.resource.data.get('teacherCommentUpdated', false) == false
            // 2026-07-06 補修：一般內容編輯同樣要求 teacherCommentContentAt 歸零，

            // 2026-06-27 補修：一般內容編輯不可夾帶變更回覆欄位

            && request.resource.data.get('studentReplyUnread', false) == resource.data.get('studentReplyUnread', false)
            && request.resource.data.get('studentReplyAt', null) == resource.data.get('studentReplyAt', null)
            && request.resource.data.get('studentReplyContentAt', null) == resource.data.get('studentReplyContentAt', null)
            // 2026-07 新增：journalSubmitNotifiedAt 語意上跟 studentReplyContentAt 同一類——

            // 任何一般編輯都不該把它動到，否則等同讓學生可以無限次把這個欄位打回 null，

                && request.resource.data.get('studentReplyContentAt', null) == resource.data.get('studentReplyContentAt', null)

    match /{path=**}/journals/{journalId} {
