rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function schoolUser() {
      return request.auth != null
        && request.auth.token.email is string
        && request.auth.token.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$');
    }
    function emailKey() {
      return request.auth.token.email
        .replace('@', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_');
    }
    function isAdmin() {
      return schoolUser()
        && (
          exists(/databases/$(database)/documents/admins/$(request.auth.uid))
          || exists(/databases/$(database)/documents/admins/$(emailKey()))
        );
    }
    function validAdminWrite() {
      return request.resource.data.email is string
        && request.resource.data.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$')
        && (
          !request.resource.data.keys().hasAny(['protected'])
          || request.resource.data.protected is bool
        );
    }
    function keepsProtectedFlag() {
      return !resource.data.keys().hasAny(['protected'])
        || resource.data.protected != true
        || (
          request.resource.data.keys().hasAny(['protected'])
          && request.resource.data.protected == true
        );
    }
    match /admins/{adminId} {
      allow get: if schoolUser() && (adminId == request.auth.uid || adminId == emailKey() || isAdmin());
      allow list: if isAdmin();
      allow create: if isAdmin() && validAdminWrite();
      allow update: if isAdmin() && validAdminWrite() && keepsProtectedFlag();
      allow delete: if isAdmin()
        && (
          !resource.data.keys().hasAny(['protected'])
          || resource.data.protected != true
        );
    }
    match /studentBindings/{bindingId} {
      allow get: if bindingId == emailKey() || isAdmin();
      allow list, create, delete: if isAdmin();
      allow update: if (
          schoolUser()
          && resource.data.email == request.auth.token.email
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['uid'])
        ) || isAdmin();
    }
    match /students/{docId} {
      // docId 格式為 {semester}_{seatNo}（新）或 {seatNo}（舊）
      // 學生可讀取自己座號的任何學期文件
      allow get: if (schoolUser() && (
          docId == get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo
          || docId.matches('.*_' + get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo)
        )) || isAdmin();
      allow list, create, update, delete: if isAdmin();
    }
    match /journals/{journalId} {
      allow read, create, update, delete: if isAdmin();
    }
    match /users/{userId} {
      allow read, write: if false;
      match /journals/{journalId} {
        allow get: if (
            schoolUser()
            && request.auth.uid == userId
            && (resource == null || resource.data.ownerUid == request.auth.uid)
          ) || isAdmin();
        allow list: if (
            schoolUser()
            && request.auth.uid == userId
          ) || isAdmin();
        allow create: if (
            schoolUser()
            && request.auth.uid == userId
            && request.resource.data.ownerUid == request.auth.uid
            && request.resource.data.ownerEmail == request.auth.token.email
            && request.resource.data.storagePath == 'user'
            // 學生不得在建立時寫入老師評語欄位（值必須為初始空值）
            && request.resource.data.get('teacherComment', null) == null
            && request.resource.data.get('teacherReviewed', false) == false
            && request.resource.data.get('reviewedAt', null) == null
            && request.resource.data.get('teacherCommentUnread', false) == false
            && request.resource.data.get('teacherCommentUpdated', false) == false
            // 學生不得在建立時寫入回覆欄位
            && request.resource.data.get('studentReply', null) == null
            && request.resource.data.get('studentReplyUnread', false) == false
          ) || isAdmin();
        allow update: if (
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.ownerUid == resource.data.ownerUid
            && request.resource.data.ownerEmail == resource.data.ownerEmail
            && request.resource.data.storagePath == 'user'
            && request.resource.data.get('teacherComment', null) == null
            && request.resource.data.get('teacherReviewed', false) == false
            && request.resource.data.get('reviewedAt', null) == null
            && request.resource.data.get('teacherCommentUnread', false) == false
            && request.resource.data.get('teacherCommentUpdated', false) == false
            // 2026-06-27 補修：一般內容編輯不可夾帶變更回覆欄位
            // （防止繞過下方第三分支的50字限制／偽造studentReplyUnread已讀狀態）
            && request.resource.data.get('studentReply', null) == resource.data.get('studentReply', null)
            && request.resource.data.get('studentReplyUnread', false) == resource.data.get('studentReplyUnread', false)
            && request.resource.data.get('studentReplyAt', null) == resource.data.get('studentReplyAt', null)
          ) || (
            // 學生將「有新評語」旗標標為已讀（只能改此單一欄位、只能設為 false）
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['teacherCommentUnread'])
            && request.resource.data.teacherCommentUnread == false
          ) || (
            // 學生回覆評語（允許 studentReply + studentReplyUnread + studentReplyAt 三個欄位）
            // 2026-06-26 更新：拿掉「舊回覆必須已被老師讀過才能再送出」的限制，
            // 改為只要老師還沒讀取（studentReplyUnread 仍是 true），學生也能持續編輯／覆蓋同一則回覆，
            // 直到老師實際讀取（saveTeacherComment 寫回 studentReplyUnread:false）為止才會換成「新一則」回覆。
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['studentReply', 'studentReplyUnread', 'studentReplyAt'])
            && request.resource.data.studentReply is string
            // 2026-06-27 補修：禁止空字串回覆（原規則 size()<=50 沒有下限，size()==0 也會通過）
            && request.resource.data.studentReply.size() >= 1
            && request.resource.data.studentReply.size() <= 50
            && request.resource.data.studentReplyUnread == true
            // 2026-06-27 補修：studentReplyAt 型別驗證（只允許 null 或字串，避免寫入任意型別的值）
            && (request.resource.data.get('studentReplyAt', null) == null || request.resource.data.studentReplyAt is string)
          ) || isAdmin();
        allow delete: if (
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
          ) || isAdmin();
      }
    }
    match /{path=**}/journals/{journalId} {
      allow read: if isAdmin();
    }
    match /deadlines/{deadlineId} {
      allow read: if schoolUser();
      allow create, update, delete: if isAdmin();
    }
    match /settings/{settingId} {
      allow read, create, update, delete: if isAdmin();
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
