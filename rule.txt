rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function schoolUser() {
      return request.auth != null
        && request.auth.token.email is string
        && request.auth.token.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$')
        // 2026-07-03 補修：要求 email_verified == true，防止透過 Email/Password
        // 自助註冊、拿一個尚未被任何 provider 使用過的校內信箱字串（例如老師已
        // 建立 studentBindings 但學生本人尚未第一次登入的空窗期）取得可通過
        // schoolUser() 的 token，進而讀取/竄改該生 binding 或冒用其座號建立月記。
        // Google 登入的真實使用者本來就是 email_verified=true，不受影響；
        // Email/Password 帳號需要真的收到並點擊驗證信才會變 true，等於要求
        // 「真的掌控該信箱」，把這條路徑拉回跟 Google OAuth 一樣「網域控制力
        // 歸屬校方」的狀態。⚠️ 部署前必須先確認 test-teacher@ / test-student@
        // 兩組測試帳號在 Firebase Console 的 email_verified 狀態為 true，否則會
        // 連 CI 自動登入一起擋下。
        && request.auth.token.email_verified == true;
    }
    function emailKey() {
      return request.auth.token.email
        .replace('@', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_')
        .replace('\\.', '_');
    }
    function isAdmin() {
      return schoolUser()
        && (
          exists(/databases/$(database)/documents/admins/$(request.auth.uid))
          || exists(/databases/$(database)/documents/admins/$(emailKey()))
        );
    }
    function validAdminWrite() {
      return request.resource.data.email is string
        && request.resource.data.email.matches('^[^@]+@tcivs\\.tc\\.edu\\.tw$')
        && (
          !request.resource.data.keys().hasAny(['protected'])
          || request.resource.data.protected is bool
        );
    }
    function keepsProtectedFlag() {
      return !resource.data.keys().hasAny(['protected'])
        || resource.data.protected != true
        || (
          request.resource.data.keys().hasAny(['protected'])
          && request.resource.data.protected == true
        );
    }
    match /admins/{adminId} {
      allow get: if schoolUser() && (adminId == request.auth.uid || adminId == emailKey() || isAdmin());
      allow list: if isAdmin();
      allow create: if isAdmin() && validAdminWrite();
      allow update: if isAdmin() && validAdminWrite() && keepsProtectedFlag();
      allow delete: if isAdmin()
        && (
          !resource.data.keys().hasAny(['protected'])
          || resource.data.protected != true
        );
      // 2026-07 新增：老師/管理員端 Web Push 用的 FCM registration token。
      // 比照 /admins/{adminId} 本身「uid 或 emailKey 皆可」的判斷方式，只有本人能寫入/刪除自己的 token；
      // 讀取只透過伺服器端腳本的 Admin SDK（notify-service/send-push-notifications.js，由
      // GitHub Actions 排程執行，非 Cloud Functions；Admin SDK 本來就會繞過此規則），
      // 前端不需要也不應該讀回自己的 token 清單。
      match /fcmTokens/{tokenId} {
        allow create, update, delete: if isAdmin() && (adminId == request.auth.uid || adminId == emailKey());
        allow get, list: if false;
      }
    }
    match /studentBindings/{bindingId} {
      // 2026-07-03 補修：原本只比對 bindingId == emailKey()，完全沒呼叫 schoolUser()，
      // 導致同一批新增的 email_verified 檢查沒套用到這裡——只要 email 字串對得上
      // 對應的 emailKey()，不論這個 token 是否真的驗證過都能讀到該筆 binding
      // （老師開學建好名冊、學生本人尚未第一次登入的空窗期正是會被利用的情境）。
      // 補上 schoolUser() 後，讀取者必須先通過「校內信箱格式 + email_verified==true」
      // 才會進一步比對 bindingId，其餘條件（bindingId 對應 emailKey()）不變。
      allow get: if (schoolUser() && bindingId == emailKey()) || isAdmin();
      allow list, create, delete: if isAdmin();
      allow update: if (
          schoolUser()
          && resource.data.email == request.auth.token.email
          && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['uid'])
        ) || isAdmin();
    }
    match /students/{docId} {
      // docId 格式為 {semester}_{seatNo}（新）或 {seatNo}（舊）
      // 學生可讀取自己座號的任何學期文件
      allow get: if (schoolUser() && (
          docId == get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo
          || docId.matches('.*_' + get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo)
        )) || isAdmin();
      allow list, create, update, delete: if isAdmin();
    }
    match /journals/{journalId} {
      allow read, create, update, delete: if isAdmin();
    }
    match /users/{userId} {
      allow read, write: if false;
      match /journals/{journalId} {
        allow get: if (
            schoolUser()
            && request.auth.uid == userId
            && (resource == null || resource.data.ownerUid == request.auth.uid)
          ) || isAdmin();
        allow list: if (
            schoolUser()
            && request.auth.uid == userId
          ) || isAdmin();
        allow create: if (
            schoolUser()
            && request.auth.uid == userId
            && request.resource.data.ownerUid == request.auth.uid
            && request.resource.data.ownerEmail == request.auth.token.email
            && request.resource.data.storagePath == 'user'
            // 2026-06-27 補修：seatNo 必須等於該學生在 studentBindings 裡記錄的座號，
            // 防止偽造 seatNo 把月記算到別的座號名下（污染老師端繳交/審閱/薪資統計）
            && request.resource.data.seatNo == get(/databases/$(database)/documents/studentBindings/$(emailKey())).data.seatNo
            // 學生不得在建立時寫入老師評語欄位（值必須為初始空值）
            && request.resource.data.get('teacherComment', null) == null
            && request.resource.data.get('teacherReviewed', false) == false
            && request.resource.data.get('reviewedAt', null) == null
            && request.resource.data.get('teacherCommentUnread', false) == false
            && request.resource.data.get('teacherCommentUpdated', false) == false
            // 2026-07-06 補修：teacherCommentContentAt 是 teacher.html saveTeacherComment()
            // 同日新增的欄位（記錄評語內容真正改變的時間，供 notify-service 判斷是否已推播），
            // 學生不得在建立時寫入，比照其餘老師專屬欄位一律限制為初始空值
            && request.resource.data.get('teacherCommentContentAt', null) == null
            // 學生不得在建立時寫入回覆欄位（studentReply / studentReplyUnread / studentReplyAt 三欄皆須為初始空值）
            && request.resource.data.get('studentReply', null) == null
            && request.resource.data.get('studentReplyUnread', false) == false
            // 2026-06-28 補修：studentReplyAt 與 studentReply／studentReplyUnread 同屬回覆欄位，
            // 建立時一律限制為 null，避免夾帶孤立時間戳記（功能上無害，但維持資料乾淨度與意圖一致性）
            && request.resource.data.get('studentReplyAt', null) == null
            // 2026-07 補修：studentReplyContentAt 與 studentReply／studentReplyUnread／studentReplyAt
            // 同屬回覆欄位家族，建立時同樣限制為初始空值（比照 teacherCommentContentAt 的設計，
            // 供 notify-service 判斷學生回覆是否已推播用途）
            && request.resource.data.get('studentReplyContentAt', null) == null
          ) || isAdmin();
        allow update: if (
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.ownerUid == resource.data.ownerUid
            && request.resource.data.ownerEmail == resource.data.ownerEmail
            && request.resource.data.storagePath == 'user'
            // 2026-06-27 補修：一般編輯不可變更 seatNo（建立時已驗證過，編輯時維持不變即可，
            // 不重新查 studentBindings——避免老師重新分配座號這種正常操作意外鎖死舊月記的編輯權）
            && request.resource.data.seatNo == resource.data.seatNo
            && request.resource.data.get('teacherComment', null) == null
            && request.resource.data.get('teacherReviewed', false) == false
            && request.resource.data.get('reviewedAt', null) == null
            && request.resource.data.get('teacherCommentUnread', false) == false
            && request.resource.data.get('teacherCommentUpdated', false) == false
            // 2026-07-06 補修：一般內容編輯同樣要求 teacherCommentContentAt 歸零，
            // 跟 reviewedAt／teacherComment 等其餘老師專屬欄位處理方式一致——內容被學生
            // 編輯後，先前『評語內容真正改變的時間』已經失去意義，理由同 create：
            // 若不歸零，理論上可被直接 API 竄改成任意值，繞過 notify-service 的已通知判斷
            && request.resource.data.get('teacherCommentContentAt', null) == null
            // 2026-06-27 補修：一般內容編輯不可夾帶變更回覆欄位
            // （防止繞過下方第三分支的50字限制／偽造studentReplyUnread已讀狀態）
            && request.resource.data.get('studentReply', null) == resource.data.get('studentReply', null)
            && request.resource.data.get('studentReplyUnread', false) == resource.data.get('studentReplyUnread', false)
            && request.resource.data.get('studentReplyAt', null) == resource.data.get('studentReplyAt', null)
            && request.resource.data.get('studentReplyContentAt', null) == resource.data.get('studentReplyContentAt', null)
          ) || (
            // 學生將「有新評語」旗標標為已讀（只能改此單一欄位、只能設為 false）
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['teacherCommentUnread'])
            && request.resource.data.teacherCommentUnread == false
          ) || (
            // 學生回覆評語（允許 studentReply + studentReplyUnread + studentReplyAt + studentReplyContentAt 四個欄位）
            // 2026-06-26 更新：拿掉「舊回覆必須已被老師讀過才能再送出」的限制，
            // 改為只要老師還沒讀取（studentReplyUnread 仍是 true），學生也能持續編輯／覆蓋同一則回覆，
            // 直到老師實際讀取（saveTeacherComment 寫回 studentReplyUnread:false）為止才會換成「新一則」回覆。
            // 2026-07 補修：比照 teacher.html saveTeacherComment() 的 commentChanged 防護——
            // 新增 studentReplyContentAt（只在回覆內容真正改變時才寫入，供 notify-service 判斷
            // 是否已推播用途）；studentReplyUnread 不再無條件要求為 true，改成依內容是否真的
            // 改變分兩種情況：內容真的改變（跟上次的 studentReply 不同）→ 必須設為 true（觸發
            // 老師端重新注意）且必須同步更新 studentReplyContentAt；內容沒變（例如老師已讀後，
            // 學生重新按送出但文字沒改）→ studentReplyUnread／studentReplyContentAt 皆須維持
            // 原值不變，不允許把老師已讀狀態誤打回未讀，也不允許時間戳前進造成 notify-service
            // 誤判為新內容而重複推播（跟 saveTeacherComment() 曾經的 isCommentUpdate 死碼／
            // commentChanged 是同一類問題，這裡從一開始就用同樣的架構處理）
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
            && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['studentReply', 'studentReplyUnread', 'studentReplyAt', 'studentReplyContentAt'])
            && request.resource.data.studentReply is string
            // 2026-06-27 補修：禁止空字串回覆（原規則 size()<=50 沒有下限，size()==0 也會通過）
            && request.resource.data.studentReply.size() >= 1
            && request.resource.data.studentReply.size() <= 50
            // 2026-06-27 補修：studentReplyAt 型別驗證（只允許 null 或字串，避免寫入任意型別的值）
            && (request.resource.data.get('studentReplyAt', null) == null || request.resource.data.studentReplyAt is string)
            && (
              (
                // 內容真的改變：必須標記未讀、且必須提供 studentReplyContentAt
                request.resource.data.studentReply != resource.data.get('studentReply', null)
                && request.resource.data.studentReplyUnread == true
                && request.resource.data.studentReplyContentAt is string
              ) || (
                // 內容沒變：studentReplyUnread／studentReplyContentAt 皆須維持原值不變
                request.resource.data.studentReply == resource.data.get('studentReply', null)
                && request.resource.data.get('studentReplyUnread', false) == resource.data.get('studentReplyUnread', false)
                && request.resource.data.get('studentReplyContentAt', null) == resource.data.get('studentReplyContentAt', null)
              )
            )
          ) || isAdmin();
        allow delete: if (
            schoolUser()
            && request.auth.uid == userId
            && resource.data.ownerUid == request.auth.uid
          ) || isAdmin();
      }
      // 2026-07 新增：學生端 Web Push 用的 FCM registration token，只有本人能寫入/刪除自己的 token；
      // 讀取只透過伺服器端腳本的 Admin SDK（notify-service/send-push-notifications.js，由
      // GitHub Actions 排程執行，非 Cloud Functions；Admin SDK 本來就會繞過此規則），
      // 前端不需要也不應該讀回自己的 token 清單。
      match /fcmTokens/{tokenId} {
        allow create, update, delete: if schoolUser() && request.auth.uid == userId;
        allow get, list: if false;
      }
    }
    match /{path=**}/journals/{journalId} {
      allow read: if isAdmin();
    }
    match /deadlines/{deadlineId} {
      allow read: if schoolUser();
      allow create, update, delete: if isAdmin();
    }
    match /settings/{settingId} {
      allow read, create, update, delete: if isAdmin();
    }
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
